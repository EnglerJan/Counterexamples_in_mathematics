\subsection{Lebesque-measurable set that is not Borel}
\par Let $\mathcal{C}\subset [0,1]$ be Cantor set, $c\:[0,1]\to[0,1]$ be cantor function defined by
\begin{align*}
	c &= \frac{1}{2} \text{ on } \left[\frac{1}{3},\frac{2}{3}\right] \\
	c &= \frac{1}{4} \text{ on } \left[\frac{1}{9},\frac{2}{9}\right], \ c=\frac{3}{4} \text{ on } \left[\frac{7}{9},\frac{8}{9}\right] \\
	  &\ \ \vdots \\
	c\left(\sum_{n\in \mathbb{N}}\frac{2a_n}{3^{n}}\right) &= \sum_{n\in \mathbb{N}}\frac{a_n}{2^n} \ \ \forall \{a_n\}_{n\in \mathbb{N}}, \ a_n\in\{0,1\} \\
	c(x) &= \sup_{y\leq x}c(y) \text{ for } x\in [0,1]\setminus \mathcal{C}
\end{align*}
Then $c$ is continuous, surjective and nondecreasing. Therefore
\begin{equation}\label{cantor+x}
  \psi(x) := c(x)+x
\end{equation}
is continuous, increasing and bijective $[0,1]\to[0,2].$ Since
\begin{align*}
	[0,1]\setminus\mathcal{C} &= \bigcup_{n\in \mathbb{N}}(a_n,b_n) \text{ where } (a_n,b_n) \text{ are intervals over which } c(x) \text{ is constant} \\
	\lambda\left(\psi\left(a_n,b_n\right)\right) &= \lambda(c(a_n)+a_n,c(b_n)+b_n)=\lambda(c|_{A_n}+a_n,c|_{A_n}+b_n)=\lambda(a_n,b_n) \\
	\Rightarrow \lambda(\psi([0,1]\setminus \mathcal{C}))&=\lambda([0,1]\setminus \mathcal{C})=1 \\
	\Rightarrow \lambda(\psi(\mathcal{C})) &= \lambda([0,2]\setminus \psi([0,1]\setminus \mathcal{C}))=1
\end{align*}
By Vitali's theorem, any Lebesque-measurable set of positive measure has a non-measurable subset. Denote 
\begin{equation}\label{nonmeasurable set}
  N\subset \psi(\mathcal{C})
\end{equation} such a non-measurable subset. Then $\psi^{-1}(N)\subset \mathcal{C}$ is Lebesque-measurable since $\lambda(\mathcal{C})=0.$ But $\psi^{-1}$ is increasing, therefore Borel, so if 
\begin{equation}\label{nonborel measurable set}
  A:= \psi^{-1}(N) 
\end{equation}
was Borel-measurable, also
\begin{equation*}
  N = (\psi^{-1})^{-1}(A)
\end{equation*}
would be, contradiction. Therefore $A$ is Lebesque-measurable but not Borel-measurable.
\vspace{.3cm} 
\par On the other hand, the completion of Borel sigma algebra is easily seen to be Lebesque sigma algebra:
\par Denote $\mathcal{B}$ Borel algebra, $\mathcal{L}$ Lebesque algebra, $\overline{\mathcal{B}}^\lambda$ the completion of $\mathcal{B}$ w.r.t. Lebesque measure $\lambda,$ $G_\delta$ to be set of countable intersections of open sets ("$G_\delta$" sets). Clearly $\overline{\mathcal{B}}^\lambda\subset \mathcal{L}.$ To prove the other inclusion, choose arbitrary $A\in \mathcal{L}.$ Then $\exists B\in G_\delta\: A\subset B, \ \lambda(B\setminus A)=0,$ so
\begin{equation*}
	A = \underbrace{B}_{\in \mathcal{B}}\setminus \underbrace{(B\setminus A)}_{\in \overline{\mathcal{B}}^\lambda, \text{ since } \lambda(B\setminus A)=0}\in \overline{\mathcal{B}}^\lambda
\end{equation*}

\subsection{Composition of Lebesque measurable functions is not Lebesque measurable}

\subsection{Hölder continuous function not absolutely continuous}
Let $c$ be the cantor function and $c_k$ be defined by
\begin{align*}
	&c_k\left(\sum_{n=1}^k \frac{2a_n}{3^n}\right)=\sum_{n=1}^k \frac{a_n}{2^n} \ \ \forall \{a_n\}_{n=0}^k, a_n\in \{0,1\} \\
	&c_k \text{ is piecewise linear on } [0,1] \\
	&c_k(0)=0, c_k(1)=1
\end{align*}
Then 
\begin{itemize}
	\item $\nm{c-c_k}_{\infty} = 2^{-k}$
	\item $c_k$ is Lipschitz with constant $\displaystyle\left(\frac{3}{2}\right)^k$ (think about the slope of steepest line)
\end{itemize}
So for $x,y \in [0,1]$
\begin{align*}
	|c(x)-c(x)|&\leq \nm{c-c_k}_{\infty} + |c_k(x)-c_k(y)| + \nm{c-c_k}_{\infty}  \\
			   &\leq 2\cdot 2^{-k} + \left(\frac{3}{2}\right)^k|x-y|
\end{align*}
Choose $k\in \mathbb{N}$ such that $3^{-k-1}\leq |x-y| < 3^{-k},$ then
\begin{equation*}
	|c(x)-c(y)|\leq 3\cdot 2^{-k} = 6\cdot 2^{-k-1} = 6\cdot 3^{-(k-1)\cdot \log_32} < 6\cdot |x-y|^{\log_32}
\end{equation*}
So the cantor function is $\log_32$-Hölder continuous. \\
Let $\epsilon>0$. Since $c$ is locally constant on $[0,1]\setminus \mathcal{C},$ which is a countable union of open intervals with measure one, by continuity of measure we can choose $x_1, ..., x_n, y_1, ..., y_n\in [0,1], x_k<y_k$ (in fact they will be endpoints of form $\frac{l}{3^n}, l,n\in \mathbb{N}$) such that
\begin{equation*}
	\sum_{k=1}^n |y_k-x_k| > 1-\epsilon \ \ \text{ and c is constant on intervals} \ \ (x_k,y_k)  
\end{equation*}
Since c is constant on big set, and $c(0)=0, c(1)=1,$ it must grow fast on a small set:
\begin{align*}
	|x_1-0| + \sum_{k=1}^{n-1} |x_{k+1}-y_{k}| + |1-y_n| = 1-\sum_{k=1}^n |y_k-x_k| &< \epsilon \\
	\text{BUT} \ \ c(x_1)-c(0) + \sum_{k=1}^{n-1}(c(x_{k+1})-c(y_k))+c(1)-c(y_n) = 1-\sum_{k=1}^n(c(y_k)-c(x_k))&=1 
\end{align*}
so $c$ is not $AC[0,1].$ 
\par A shorter proof might use characterization $AC[0,1]=W^{1,1}(0,1),$ since $c$ has classical derivative zero almost everywhere, but certainly
\begin{equation*}
	c(0)+\int_{0}^1 c'(x)dx = 0+\int_{0}^1 0\ dx = 0 \neq c(1)
\end{equation*}
In fact, $c\in BV[0,1]\setminus AC[0,1]$ and it's Lebesque-Stieltjes measure is singular nonatomic with respect to the Lebesque measure on $[0,1].$

\subsection{Fréchet space that is not normable}
Consider $L^1_{loc}(\mathbb{R}^d)$ given the Fréchet topology generated by seminorms
\begin{equation*}
	|f|_n = \int_{\nm{x}\leq n}|f(y)|dy
\end{equation*}
Suppose there is a norm $\nm{\bullet}$ generating the same topology. Then $\exists C>0, C_n>0, n\in \mathbb{N}$ such that
\begin{align}
	\nm{\bullet}&\leq C|\!\bullet\!|_N, \label{nm<nmN}\\
	|\!\bullet\!|_n &\leq C_n \nm{\bullet}
\end{align}
from this we get $|\!\bullet\!|_{N+1}\leq C_{N+1}C|\!\bullet\!|_N$, which is clearly wrong for the function $f=\chi_{\{N<\nm{x}<N+1\}}.$
\par To prove (\ref{nm<nmN}), for contradiction, assume
\begin{equation*}
	\forall n,k\in \mathbb{N} \ \exists \phi_{n,k}\in L^{1}_{loc}: \nm{\phi_{n,k}}>n|\phi_{n,k}|_k
\end{equation*}
define $\psi_n = \displaystyle\frac{\phi_{n,n}}{n|\phi_{n,n}|_n},$ then 
\begin{equation*}
  \forall l\in \mathbb{N}, n\geq l: |\psi_n|_l \leq |\psi_n|_n = \frac{1}{n} \to 0 \ \ \ \ n\to \infty
\end{equation*}
so $\psi_n \to 0$ in $L^1_{loc}$ but $\nm{\psi_n}>1,$ contradiction. Generally, for any Fréchet space $X$ where the seminorms are nondecreasing, the inequalities (1) and (2) hold, so if we can find elements $x\in X$ such that $|x|_{N+1}$ can be arbitrarily large and $|x|_N$ arbitrarily small, the space is nor normable.
\par For $C^\infty_0[0,1]$ where topology is generated by $|f|_n = \sup_{k\leq n}\nm{f^{(k)}}_\infty$, the element $f(x)=\frac{1}{n^N}\sin(nx)$ ($n$ large) has small $|\bullet|_{N-1}$ but large $|\bullet|_N.$

\subsection{Momentum operator on half-line does not exist}
Let $A\: L^{2}(0,\infty)\to L^{2}(0,\infty)$ be 
\begin{equation*}
	Au := iu', \ \ D(A):=\{u\in W^{1,2}(0,\infty)\: u(0)=0\}
\end{equation*}
To show $A$ is symmetric, in the following computation 
\begin{equation*}
	\forall u,\varphi\in D(A)\: (Au,\varphi) = \int_0^\infty iu'\overline{\varphi}\ dx = \int_0^\infty u\overline{i\varphi'}\ dx+i[u\overline{\varphi}]_0^\infty = (u,A\varphi)+i (u\overline{\varphi})\rvert_{\infty}
\end{equation*}
we need that $\displaystyle\lim_{x\to \infty} u(x)\overline{\varphi(x)}=0.$ We show that for all $v\in W^{1,2}(0,\infty),\ \displaystyle\lim_{x\to \infty}v(x)=0.$
\begin{equation*}
	\forall y<x\in \mathbb{R}\: v^{2}(x)-v^{2}(y)=\int_y^x 2v(t)v'(t)dt \leq 2\nm{v}_{L^{2}(y,\infty)}\nm{v'}_{L^{2}(0,\infty)}\stackrel{y\to \infty}{\to} 0
\end{equation*}
therefore $v(y)$ is Cauchy for $y\to \infty$, and so $\displaystyle\lim_{y\to \infty}v(y)$ exists. From $v\in L^{2}(0,\infty)$ it follows $\displaystyle\lim_{y\to \infty}v(y)=0.$ Therefore $A$ is symmetric.
To calculate the defect indices
\begin{align*}
	n_{+}(A)&:= \dim \Im(A+i \mathbb{1})^\perp=\dim \Ker(A^{*}-i \mathbb{1}),\\  n_{-}(A)&:= \dim \Im(A-i \mathbb{1})^\perp=\dim \Ker(A^{*}+i \mathbb{1})
\end{align*}
we need to calculate $A^{*}$:
\begin{align*}
	D(A^{*})&=\{\varphi\in L^{2}(0,\infty)\: \exists \psi\in L^{2}(0,\infty)\ \forall u\in D(A)\: (Au,\varphi)=(u,\psi)\} \\
	\forall u\in D&(A)\: (Au,\varphi)=\int_0^\infty iu'\overline{\varphi}\ dx = \underbrace{i[u\overline{\varphi}]_0^\infty}_{=0} + \int_0^\infty u\overline{i\varphi'}dx \ \Rightarrow \ D(A^{*})=W^{1,2}(0,\infty) \\
	n_+(A)&=\dim \{u\in W^{1,2}(0,\infty)\: iu'-iu=0\}=\dim (\Span\{e^{x}\}\cap W^{1,2}(0,\infty))=0 \\
	n_-(A)&=\dim \{u\in W^{1,2}(0,\infty)\: iu'+iu=0\}=\dim (\Span\{e^{-x}\}\cap W^{1,2}(0,\infty))=1 \\
\end{align*}
Since $n_+ \neq n_-,$ there is no self-adjoint extension.

\subsection{Fourier series of continuous function does not converge}.
Let for $f\: [0,2\pi]\to \mathbb{R}$
\begin{equation*}
	S_Nf(x)=\sum_{n=-N}^Na_ne^{-inx} = \frac{1}{2\pi}\int_0^{2\pi}D_N(x-t)f(t)dt,
\end{equation*}
where $D_N$ is the Dirichlet kernel:
\begin{equation*}
  D_N(x)=\frac{\sin\left(\frac{2N+1}{2}x\right)}{\sin\left(\frac{x}{2}\right)}.
\end{equation*}
It is well-known that 
\begin{itemize}
	\item for $f\in L^{p}(0,2\pi),$ $S_Nf \to f$ in $L^{p}(0,2\pi), 1<p<\infty,$
	\item for $f\in BV[0,2\pi],$ $S_Nf(x)\to \lim_{\epsilon\to 0}\frac{f(x+\epsilon)+f(x-\epsilon)}{2}$ for $x\in[0,2\pi],$ 
	\item for $f\in BV[0,2\pi]\cap C[0,2\pi],$ $S_Nf\rightrightarrows f$ on $[0,2\pi].$ 
\end{itemize}
Fix $x\in [0,2\pi], N\in \mathbb{N}.$ Then define a signed Radon measure
\begin{equation*}
	\Phi_{x,N}\: C[0,2\pi]\to \mathbb{R}, \ \ \ \Phi_{x,N}f := S_Nf(x)
\end{equation*}
it's norm is
\begin{equation*}
	||\Phi_{x,N}||_{(C[0,2\pi])^{*}}=\frac{1}{2\pi}\int_0^{2\pi}|D_N(s)|dt
\end{equation*}
But
\begin{equation*}
	\frac{1}{2\pi}\int_0^{2\pi}|D_N(s)|dt \geq \frac{1}{2\pi}\int_0^{2\pi} \frac{|\sin\left(\frac{2N+1}{2}s\right)|}{\frac{s}{2}}ds = \int_0^{(2N+1)\pi}\frac{|\sin x|}{x}dx \stackrel{N\to \infty}{\to} \int_0^{\infty}\frac{|\sin x|}{x}dx=\infty
\end{equation*}
So $\{\Phi_{x,N}\}_{N\in \mathbb{N}}$ is not bounded in $(C[0,2\pi])^{*}.$ By Uniform Boundedness Principle we get 
\begin{equation}
	\forall x\in [0,2\pi] \ \exists \text{ dense set } \mathcal{A}\subset C[0,2\pi] \text{ such that } \forall f\in \mathcal{A}\: |S_Nf(x)|\stackrel{N\to\infty}{\to} \infty
\end{equation}

\subsection{Functional not attaining it's norm. Counterexample to projection theorem}
Let $X=L^{1}(0,1),\ \phi\in X^{*}$ defined by $\langle\phi,f\rangle:= \int_0^1 xf(x)dx.$ We will show
\begin{equation*}
	\nexists f\in X\: \nm{f}_X=1, \ |\langle \phi, f\rangle|=\nm{\phi}_{X^{*}}.
\end{equation*}
Clearly 
\begin{equation*}
  \nm{\phi}_{(L^{1}(0,1))^{*}}=\nm{x}_{L^{\infty}(0,1)}=1. 
\end{equation*}
For contradiction, suppose there exist $f\in L^{1}(0,1)$ such that $\nm{f}_1=1$, $|\langle \phi, f\rangle|=1.$ Then $f$ does not change sign on $(0,1)$ so we can choose $f$ such that $f\geq 0$ on $(0,1)$ (if not, then $\nm{|f|}_{L^{1}(0,1)}=\nm{f}_{L^{1}(0,1)}=1, \ |\langle \phi, |f|\rangle|>|\langle \phi, f\rangle|=1,$ contradicting $\nm{\phi}_{(L^{1})^{*}}=1$). Then
\begin{align*}
1
&= \int_0^{1-\frac{1}{n}}xf(x)\,dx+\int_{1-\frac{1}{n}}^1xf(x)\,dx \leq \\
&\leq \left(1-\frac{1}{n}\right)\int_0^{1-\frac{1}{n}}f(x)\,dx + \int_{1-\frac{1}{n}}^1f(x)\,dx \\
&= \left(1-\frac{1}{n}\right)\underbrace{\int_0^1f(x)\,dx}_{=1}
+ \frac{1}{n}\int_{1-\frac{1}{n}}^1f(x)\,dx
\end{align*}
\vspace{-.8cm}
\begin{alignat*}{2}
&\Rightarrow \quad &\int_{1-\frac{1}{n}}^1 f(x)\,dx &\ge 1 \qquad \forall n\in\mathbb N \\
& & &\ \big\downarrow \; n\to\infty \quad (f\in L^{1}) \\
& & 0 &\ge 1.
\end{alignat*}
Intuitively, any $f$ attaining the bound $|\langle \phi, f\rangle|=1$ would have to have all it's mass at $x=1,$ since if the mass is anywhere else, we can move it closer to $x=1$ to make $\langle \phi,f\rangle$ bigger. This is impossible on the level of $L^{1}$ functions. However, taking $\delta_1 \in (C[0,1])^{*}, \ \langle\delta_1,\varphi \rangle=\varphi(1),$ by Hahn-Banach
\begin{equation*}
	\nm{\delta_1}_{(C[0,1])^{*}}=1, \ C[0,1]<L^{\infty}(0,1) \Rightarrow \exists T\in (L^{\infty}(0,1))^{*}\: T\rvert_{C[0,1]}=\delta_1, \ \nm{T}_{(L^{\infty}(0,1))^{*}}=1.
\end{equation*}
And (equating $\phi\in (L^{1})^{*}$ with $x\in L^{\infty}$):
\begin{equation*}
  \langle T, \phi \rangle = \langle T, x \rangle = \langle \delta_1, x \rangle=1.
\end{equation*}
Therefore $\phi$ attains it's maximum on the unit ball of dual $(L^{\infty})^{*}$, but not on predual $L^{1}$. The problem is lack of reflexivity. By James' theorem, for any Banach space $Y$
\begin{center}
  $Y$ is reflexive $\iff$ any $\phi\in Y^{*}$ attains it's norm on the unit ball $B_1(0)\subset Y.$
\end{center}
We say that a subspace $A\subset Y$ of Banach space is \textit{proximital}, if
\begin{equation*}
	\forall x\in Y \ \exists! a\in A\: \nm{x-a}=\inf_{b\in A}\nm{x-b} \ \ \ \hdots a:=P_A(x)
\end{equation*}
by the Projection theorem, every nonempty closed convex subset of a Hilbert space is proximital. We show that in $X=L^{1}$, $V:=\{f\in L^{1}(0,1)\: \langle \phi, f\rangle=1\}$ is not proximital, even though it is a closed convex subset. We wish to project $0\in L^{1}$ to $V.$ Clearly
\begin{equation*}
	\forall \epsilon>0\ \exists f_\epsilon\in L^{1}\: \nm{f_\epsilon}_{L^{1}}\leq 1+\epsilon , \ \langle \phi, f_\epsilon\rangle=1 \ \ \ \&\ \ \ \forall f\in L^{1}\: \langle \phi, f\rangle=1 \Rightarrow \nm{f}_{L^{1}}\geq 1
\end{equation*}
\begin{equation*}
	\Rightarrow  \inf_{f\in V}\nm{0-f} = \inf_{f\in V}\nm{f} = 1
\end{equation*}
But we know that this bound cannot be attained, so the projection does not exist:
\begin{equation*}
	f= P_V(0)\ \Rightarrow \ \nm{f}_{L^{1}}=1\ \ \& \ \ f\in V, \ \ \ \text{a contradiction.}
\end{equation*}
